from pathlib import Path
import subprocess, os, json
branch='codex/brief-0-5-grader-and-validators'
docs=['docs/audits/BACKLOG.md','docs/audits/brief-0-5/EVIDENCE.md','next-env.d.ts']
work={p:Path(p).read_bytes() for p in docs}
results=[]
root=Path('docs/audits/brief-0-5/evidence/browser-comparison');root.mkdir(parents=True,exist_ok=True)
cmd=['node','node_modules/@playwright/test/cli.js','test','e2e/calendar-journeys.spec.ts','e2e/calendar-recurring.spec.ts','e2e/calendar-tab.spec.ts','e2e/core-learning-loop.spec.ts','--grep',"two simultaneous tabs|concurrent edits to a recurring series|finishing a quick-add plan|a confident misconception|a new topic is taught",'--config','playwright.brief05-verification.config.ts']
def run(label):
 out=(root/label).resolve();out.mkdir(parents=True,exist_ok=True)
 env=dict(os.environ,BRIEF05_BROWSER_OUTPUT=str(out))
 with (out/'run.txt').open('w') as log:r=subprocess.run(cmd,env=env,stdout=log,stderr=subprocess.STDOUT)
 results.append({'revision':subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip(),'label':label,'exitCode':r.returncode})
 print(label,r.returncode,flush=True)
try:
 for p in docs:Path(p).write_bytes(subprocess.check_output(['git','show','HEAD:'+p]))
 subprocess.run(['git','switch','--detach','e03a082'],check=True)
 assert not subprocess.check_output(['git','diff','--name-only','e03a082','--','src'],text=True).strip()
 run('main')
finally:
 Path('next-env.d.ts').write_bytes(subprocess.check_output(['git','show','HEAD:next-env.d.ts']))
 subprocess.run(['git','switch',branch],check=True)
 for p,data in work.items():Path(p).write_bytes(data)
 print('Restored Brief 0.5 branch and audit notes.',flush=True)
run('branch')
(root/'manifest.json').write_text(json.dumps({'serial':True,'runnerRetries':0,'command':cmd,'results':results},indent=2)+'\n')
